# timestomp_detection.ps1 — ตรวจ timestomping บน Windows / XAMPP
# Detect timestomping โดยเปรียบเทียบ LastWriteTime (mtime) กับ CreationTime
# ถ้า LastWriteTime < CreationTime = timestamp ถูกปลอม (ปกติ LastWriteTime ต้อง >= CreationTime)

$root = "C:\xampp\htdocs"
Get-ChildItem -Path $root -Recurse -File -Include *.gif,*.php,*.phtml,*.htaccess |
    Where-Object { $_.LastWriteTime -lt $_.CreationTime } |
    ForEach-Object {
        Write-Host ("TIMESTOMPED: {0}" -f $_.FullName)
        Write-Host ("  LastWriteTime: {0} (forged)" -f $_.LastWriteTime)
        Write-Host ("  CreationTime : {0} (actual)" -f $_.CreationTime)
    }
